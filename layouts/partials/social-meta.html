{{/* Open Graph and Twitter Card tags so shared links show a title, description, and featured image. Featured images in a page bundle are resized to a 1200px-wide JPEG (what social crawlers handle best); older posts keep their /uploads/... path. Pages without a featured image fall back to the avatar. */}} {{ $title := cond .IsHome site.Title .Title }} {{ $desc := "" }} {{ with .Description }}{{ $desc = . }}{{ else }}{{ if and (not .IsHome) .Summary }}{{ $desc = .Summary | plainify | truncate 200 }}{{ else }}{{ $desc = site.Params.description }}{{ end }}{{ end }} {{ $image := "" }} {{ $w := 0 }} {{ $h := 0 }} {{ with .Params.featured_image }} {{ with $.Resources.GetMatch . }} {{ if eq .ResourceType "image" }} {{ $img := .Process "resize 1200x jpg q85" }} {{ $image = $img.Permalink }} {{ $w = $img.Width }} {{ $h = $img.Height }} {{ end }} {{ else }} {{ $image = . | absURL }} {{ end }} {{ end }} {{ $large := ne $image "" }} {{ if not $large }}{{ $image = "images/avatar.png" | absURL }}{{ end }} {{ if and $large (gt $w 0) }} {{ end }} {{ if eq .Section "posts" }} {{ end }}
